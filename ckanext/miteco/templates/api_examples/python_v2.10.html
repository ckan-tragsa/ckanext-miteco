  import urllib.request
  import urllib.parse

  base_url = '{{ h.url_for('api.action', ver=3, logic_function='datastore_search', qualified=True) }}'
  params = {
    'resource_id': '{{resource_id}}', # the resource id
    'limit': 5, # get 5 results
    'q': 'medio-ambiente' # query for 'medio-ambiente'
  }

  query_string = urllib.parse.urlencode(params)
  url = base_url + '?' + query_string

  try:
    fileobj = urllib.request.urlopen(url)
    print(fileobj.read().decode('utf-8'))
  except urllib.error.URLError as e:
    print(f"Error al contactar la URL: {e.reason}")
